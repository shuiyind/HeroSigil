```java
private void drawBuffIcon(GuiGraphics guiGraphics, PoseStack poseStack) {
    Minecraft minecraft = Minecraft.getInstance();
    float centerX = (float) (this.getX() + this.getWidth() / 2.0);
    float centerY = (float) (this.getY() + this.getHeight() / 2.0);
    float radius = Math.min(this.getWidth(), this.getHeight()) / 2.0f - 2.0f;

    // 圆形背景
    guiGraphics.fillCircle(centerX, centerY, radius + 1, 0xFF4682B4); // 钢蓝色外环
    guiGraphics.fillCircle(centerX, centerY, radius, 0xFF1E3A5F); // 深蓝内部

    // 获取 Buff 类型并绘制图标
    BuffType type = this.achievement.getBuffType();
    char firstLetter = type.getFirstLetter();

    if (minecraft.font.width(String.valueOf(firstLetter)) > radius * 2 - 4) {
        float scale = (radius * 2 - 4) / (float) minecraft.font.width(String.valueOf(firstLetter));
        poseStack.pushPose();
        poseStack.scale(scale, scale, scale);
        guiGraphics.drawString(minecraft.font, 
            String.valueOf(firstLetter), 
            (int) (centerX - minecraft.font.width(String.valueOf(firstLetter)) / 2.0 / scale), 
            (int) (centerY - minecraft.font.height("") / 2.0 / scale + 1), 
            0xFFFFFFFF, false);
        poseStack.popPose();
    } else {
        guiGraphics.drawString(minecraft.font, 
            firstLetter.isEmpty() ? "?" : String.valueOf(firstLetter), 
            (int) (centerX - minecraft.font.width(firstLetter.isEmpty() ? "?" : String.valueOf(firstLetter)) / 2.0), 
            (int) (centerY - minecraft.font.height("") / 2.0 + 1), 
            0xFFFFFFFF, false);
    }
}
```

```java
/**
 * 根据 Buff 类型绘制专属图标
 */
private void drawBuffTypeIcon(GuiGraphics guiGraphics, PoseStack poseStack, 
                               float centerX, float centerY, float radius, 
                               BuffType type) {
    switch (type) {
        case HEALTH_BOOST:
            drawHeartIcon(guiGraphics, poseStack, centerX, centerY, radius);
            break;
        case SPEED_BOOST:
            drawLightningIcon(guiGraphics, poseStack, centerX, centerY, radius);
            break;
        case SATURATION:
        case REGENERATION:
            drawShieldIcon(guiGraphics, poseStack, centerX, centerY, radius);
            break;
        default:
            // 默认绘制首字母
            char firstLetter = type.getFirstLetter();
            guiGraphics.drawString(Minecraft.getInstance().font,
                String.valueOf(firstLetter),
                (int) (centerX - Minecraft.getInstance().font.width(String.valueOf(firstLetter)) / 2.0),
                (int) (centerY - Minecraft.getInstance().font.height("") / 2.0 + 1),
                0xFFFFFFFF, false);
    }
}
```

```java
private void drawHeartIcon(GuiGraphics guiGraphics, PoseStack poseStack,
                           float centerX, float centerY, float radius) {
    // 简化的心形几何图形
    float size = radius * 0.7f;
    
    // 心形由两个圆形和一个三角形组成
    int heartColor = 0xFFE74C3C; // 红色

    guiGraphics.fillCircle(centerX - size * 0.3f, centerY - size * 0.3f, size * 0.5f, heartColor);
    // 右上半圆
    guiGraphics.fillCircle(centerX + size * 0.3f, centerY - size * 0.3f, size * 0.5f, heartColor);
    // 下半部分 (三角形)
    guiGraphics.fillTriangle(centerX - size * 0.5f, centerY - size * 0.1f,
                            centerX + size * 0.5f, centerY - size * 0.1f,
                            centerX, centerY + size * 0.6f, heartColor);
}
```
